# 3. CONTAINER START COMMAND
# ------------------------------------------------------------------------------
# Paste the contents of runpod-start.json into the "Container Start Command" field
# (the readable version is bootstrap.sh).
# What it does: checks DNS for github.com + huggingface.co (falls back to
# 1.1.1.1 / 8.8.8.8 if needed), clones this repo (10 retries), runs setup.sh.
# setup.sh installs everything, downloads the models and finally runs /start.sh.

# 4. STORAGE

Container disk:           5 GB     # disposable system files only
Volume disk:              0 GB     # do NOT use a pod-local volume disk
Volume mount path:        /workspace

# At deploy time, each user attaches their OWN Network Volume at /workspace.

# setup.sh refuses to run if /workspace is the container disk or a Global Volume.
# Storage types: https://docs.runpod.io/pods/storage/types

# 8. WHAT GETS LOADED ON FIRST BOOT (automatic, ~13.6 GB)
# ------------------------------------------------------------------------------
# Custom nodes
#   comfyui-qwen3-tts (bundled in this repo, custom_nodes/)
#   Python packages added to ComfyUI's venv: qwen-tts==0.1.1 (no deps),
#   transformers==4.57.3, accelerate==1.12.0, librosa, soundfile, sox, onnxruntime,
#   einops. PyTorch stays the image's.
#     https://github.com/QwenLM/Qwen3-TTS

# Models, under ComfyUI/models/qwen3-tts/ (each: model, text tokenizer, 12 Hz speech
# tokenizer; all Apache-2.0)
#   Qwen3-TTS-12Hz-1.7B-CustomVoice/   ~4.5 GB   the 9 built-in voices
#     https://huggingface.co/Qwen/Qwen3-TTS-12Hz-1.7B-CustomVoice
#   Qwen3-TTS-12Hz-1.7B-VoiceDesign/   ~4.5 GB   a voice from a description
#     https://huggingface.co/Qwen/Qwen3-TTS-12Hz-1.7B-VoiceDesign
#   Qwen3-TTS-12Hz-1.7B-Base/          ~4.5 GB   voice cloning
#     https://huggingface.co/Qwen/Qwen3-TTS-12Hz-1.7B-Base

# Voice pool (copied into ComfyUI/input/voices/, never over an existing file)
#   16 clips + transcripts: Ryan, Aiden (English); Vivian, Serena, Uncle_Fu, Dylan,
#   Eric (Chinese), Ono_Anna (Japanese), Sohee (Korean), each also as <Name>_en
#   (English). Rendered once and shipped in samples/voices/, so no GPU run is needed.
#   Also input/example_recording.flac (a copy of Serena_en) for workflow 4.

# Workflows (copied into ComfyUI's Workflows sidebar; edited copies are never overwritten)
#   qwen3tts_1_builtin_voice     a built-in voice reads your script (any length)
#   qwen3tts_2_voice_design      describe a voice; its sample goes to the voice pool
#   qwen3tts_3_voice_clone       any pool voice reads your script, the same throughout
#   qwen3tts_4_add_your_voice    a recording + transcript into the voice pool
# Voiceovers: ComfyUI/output/voiceover/<prefix>_00001.wav and .mp3


# ------------------------------------------------------------------------------
